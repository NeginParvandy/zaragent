class ChatRequest(BaseModel):
    """
    Request ط§طµظ„غŒ Agent Chat.
    """

    model_config = ConfigDict(
        populate_by_name=True,
        extra="ignore",
    )

    message: str = Field(
        min_length=1,
        max_length=2000,
    )

    context: ChatContextRequest | None = None

    @field_validator("message")
    @classmethod
    def message_not_empty(
        cls,
        value: str,
    ) -> str:
        value = (value or "").strip()

        if not value:
            raise ValueError(
                "message is required"
            )

        return value
